! ============================================================
!  Router: R1 — GRE Hub (HQ Site, 192.168.0.0/24)
!  Device: Cisco 2911
!  Role:   Establishes two GRE tunnels:
!          Tunnel116 → R2 (BR1, 192.168.1.0/24)
!          Tunnel117 → R3 (BR2, 192.168.2.0/24)
!  Note:   Passwords removed for publication.
! ============================================================

hostname R1
!
enable secret <removed>
!
! --- GRE Tunnel to R2 (BR1) ---
interface Tunnel116
 ip address 172.16.0.10 255.255.255.0
 mtu 1476
 tunnel source Serial0/0/0
 tunnel destination 2.2.2.2
!
! --- GRE Tunnel to R3 (BR2) ---
interface Tunnel117
 ip address 172.17.0.10 255.255.255.0
 mtu 1476
 tunnel source Serial0/0/0
 tunnel destination 3.3.3.3
!
interface GigabitEthernet0/0
 description LAN — HQ Site (192.168.0.0/24)
 ip address 192.168.0.1 255.255.255.0
 duplex auto
 speed auto
!
interface Serial0/0/0
 description WAN — Uplink to ISP (1.1.1.0/24)
 ip address 1.1.1.1 255.255.255.0
!
ip classless
! Routes to remote WAN endpoints (via ISP)
ip route 2.2.2.0 255.255.255.0 1.1.1.2
ip route 3.3.3.0 255.255.255.0 1.1.1.2
! Routes to branch LANs (via GRE tunnels)
ip route 192.168.1.0 255.255.255.0 172.16.0.11
ip route 192.168.2.0 255.255.255.0 172.17.0.11
!
line vty 0 4
 login
!
end
